cmake_minimum_required(VERSION 3.18)
project(easydict_speech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(RH ${CMAKE_CURRENT_SOURCE_DIR}/vendor/rhvoice/src)
file(GLOB CORE ${RH}/core/*.cpp)
list(FILTER CORE EXCLUDE REGEX "/(emoji_data|unidata|lts|russian|english|esperanto|georgian|ukrainian|kyrgyz|tatar|brazilian_portuguese|macedonian|vietnamese)\\.cpp$")
file(GLOB HTS ${RH}/hts_engine/HTS_*.c)
# CONFIGURE leaves unchanged files alone; rewriting them would recompile a third of the library on every build.
file(CONFIGURE OUTPUT config.h CONTENT "#pragma once\nconst char CONFIG_PATH[] = \"\";\nconst char DATA_PATH[] = \"\";\n" @ONLY)
file(CONFIGURE OUTPUT core/config.h CONTENT "#pragma once\n#define ENABLE_SONIC 0\n#define ENABLE_PKG 0\nconst char VERSION[] = \"1.18.4\";\n" @ONLY)
set(ES ${CMAKE_CURRENT_SOURCE_DIR}/vendor/espeak-ng/src)
file(GLOB ESPEAK ${ES}/libespeak-ng/*.c ${ES}/ucd-tools/src/*.c)
file(CONFIGURE OUTPUT espeak/config.h CONTENT "#define PACKAGE_VERSION \"1.52.0\"\n#define HAVE_MKSTEMP !_WIN32\n" @ONLY)
add_library(espeak STATIC ${ESPEAK})
set_target_properties(espeak PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 99)
target_include_directories(espeak PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/espeak ${ES}/libespeak-ng ${ES}/include ${ES}/include/compat ${ES}/ucd-tools/src/include)
# eSpeak's default of 160 bytes is too short for the data folder of an app unpacked in a deep macOS folder.
target_compile_definitions(espeak PRIVATE LIBESPEAK_NG_EXPORT=1 N_PATH_HOME=1024 PATH_ESPEAK_DATA="")
add_library(easydict_speech SHARED speech.cpp vits.cpp miniaudio.c ${CORE} ${HTS} ${RH}/lib/lib.cpp)
target_include_directories(easydict_speech PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ${RH}/include ${RH}/hts_engine ${RH}/third-party/utf8 ${RH}/third-party/rapidxml ${ES}/include)
target_compile_definitions(easydict_speech PRIVATE _LIBCPP_ENABLE_CXX17_REMOVED_UNARY_BINARY_FUNCTION LIBESPEAK_NG_EXPORT=1)
# The convolutions are nearly the whole cost of a word, and only vectorised do they keep up with a tap.
if(MSVC)
  # /RTC1, which Debug brings, refuses /O2.
  set_source_files_properties(vits.cpp PROPERTIES COMPILE_OPTIONS "$<$<NOT:$<CONFIG:Debug>>:/O2>;/fp:fast")
  target_compile_options(espeak PRIVATE /utf-8)
  target_compile_definitions(espeak PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
  target_compile_options(easydict_speech PRIVATE /utf-8 /bigobj)
  # RHVoice derives from std::unary_function, which C++17 removed from MSVC's STL.
  target_compile_definitions(easydict_speech PRIVATE WIN32 UNICODE _UNICODE NOMINMAX _HAS_AUTO_PTR_ETC=1)
else()
  set_source_files_properties(vits.cpp PROPERTIES COMPILE_OPTIONS "-O3;-ffast-math")
  target_compile_options(espeak PRIVATE -fvisibility=hidden -fwrapv)
  target_compile_options(easydict_speech PRIVATE -fvisibility=hidden)
endif()
find_package(Threads REQUIRED)
target_link_libraries(easydict_speech PRIVATE espeak Threads::Threads ${CMAKE_DL_LIBS})
option(SPEECH_CLI "Build the server WAV renderer" OFF)
if(SPEECH_CLI)
  add_executable(easydict-speech cli.cpp)
  target_link_libraries(easydict-speech PRIVATE easydict_speech)
  set_target_properties(easydict-speech PROPERTIES BUILD_RPATH_USE_ORIGIN TRUE)
endif()
